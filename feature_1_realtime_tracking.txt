ФИЧА 1: REAL-TIME DRONE TRACKING & FLEET MANAGEMENT DASHBOARD
============================================================================

ОПИСАНИЕ ФУНКЦИИ
================

Система отслеживания дронов в реальном времени с интерактивным дашбордом для управления флотом. Обеспечивает мониторинг всех активных дронов, их статус, телеметрию и местоположение в режиме реального времени с возможностью оперативного реагирования на критические ситуации.

ОСНОВНЫЕ КОМПОНЕНТЫ
==================

1. WebSocket Server для real-time коммуникации
2. Интерактивная карта с позициями дронов  
3. Дашборд с телеметрией и статистикой
4. Система алертов и уведомлений
5. Fleet analytics модуль
6. Live командная панель
7. Геозоны и ограничения

ТЕХНИЧЕСКИЕ ТРЕБОВАНИЯ
======================

Backend:
- WebSocket сервер (socket.io)
- Redis для кэширования состояний дронов
- MongoDB для хранения исторических данных
- Event-driven архитектура
- Геопространственные индексы

Frontend:
- React/Vue.js дашборд
- Leaflet/Mapbox для карт
- Chart.js для графиков
- Socket.io-client для real-time
- WebRTC для видео стримов (опционально)

1.3. WebSocket Server Implementation:
```javascript
// src/websocket/socketServer.js
import { Server } from 'socket.io';
import Redis from 'ioredis';
import { authenticateSocket } from './middleware/socketAuth.js';
import DroneHandler from './handlers/droneHandler.js';
import FleetHandler from './handlers/fleetHandler.js';

class WebSocketServer {
  constructor(httpServer) {
    this.io = new Server(httpServer, {
      cors: {
        origin: process.env.FRONTEND_URL || "http://localhost:3000",
        credentials: true
      },
      path: '/api/ws'
    });

    this.redis = new Redis({
      host: process.env.REDIS_HOST || 'localhost',
      port: process.env.REDIS_PORT || 6379,
      password: process.env.REDIS_PASSWORD
    });

    this.activeDrones = new Map();
    this.userSessions = new Map();
    
    this.setupMiddleware();
    this.setupEventHandlers();
  }

  setupMiddleware() {
    this.io.use(authenticateSocket);
    
    this.io.use((socket, next) => {
      // Rate limiting
      const userId = socket.userId;
      const now = Date.now();
      const windowMs = 60000; // 1 minute
      const maxRequests = 100;

      // Implement rate limiting logic
      next();
    });
  }

  setupEventHandlers() {
    this.io.on('connection', (socket) => {
      console.log(`User ${socket.userId} connected`);
      
      // Store user session
      this.userSessions.set(socket.userId, {
        socketId: socket.id,
        connectedAt: new Date(),
        subscribedDrones: new Set()
      });

      // Join user to personal room
      socket.join(`user:${socket.userId}`);

      // Setup drone handlers
      this.setupDroneHandlers(socket);
      this.setupFleetHandlers(socket);
      this.setupCommandHandlers(socket);

      socket.on('disconnect', () => {
        console.log(`User ${socket.userId} disconnected`);
        this.userSessions.delete(socket.userId);
      });
    });
  }

  setupDroneHandlers(socket) {
    // Subscribe to drone updates
    socket.on('drone:subscribe', async (droneId) => {
      const userSession = this.userSessions.get(socket.userId);
      if (userSession) {
        userSession.subscribedDrones.add(droneId);
        socket.join(`drone:${droneId}`);
        
        // Send current drone state
        const droneState = await this.getDroneState(droneId);
        socket.emit('drone:state_update', droneState);
      }
    });

    socket.on('drone:unsubscribe', (droneId) => {
      const userSession = this.userSessions.get(socket.userId);
      if (userSession) {
        userSession.subscribedDrones.delete(droneId);
        socket.leave(`drone:${droneId}`);
      }
    });

    // Handle telemetry updates from drones
    socket.on('drone:telemetry', async (data) => {
      await this.processTelemetry(data, socket.userId);
    });
  }

  setupFleetHandlers(socket) {
    // Subscribe to fleet updates
    socket.on('fleet:subscribe', () => {
      socket.join('fleet:updates');
      this.sendFleetOverview(socket);
    });

    socket.on('fleet:unsubscribe', () => {
      socket.leave('fleet:updates');
    });
  }

  setupCommandHandlers(socket) {
    // Send command to drone
    socket.on('drone:command', async (data) => {
      const { droneId, command, parameters } = data;
      await this.sendDroneCommand(droneId, command, parameters, socket.userId);
    });

    // Emergency stop
    socket.on('drone:emergency_stop', async (droneId) => {
      await this.emergencyStop(droneId, socket.userId);
    });
  }

  async processTelemetry(telemetryData, userId) {
    const { droneId, ...telemetry } = telemetryData;
    
    // Validate telemetry data
    if (!this.validateTelemetry(telemetry)) {
      return;
    }

    // Store in Redis for real-time access
    await this.redis.setex(
      `drone:${droneId}:telemetry`, 
      300, // 5 minutes TTL
      JSON.stringify({
        ...telemetry,
        timestamp: new Date(),
        userId
      })
    );

    // Update active drones map
    this.activeDrones.set(droneId, {
      ...telemetry,
      lastUpdate: new Date(),
      userId
    });

    // Check for alerts
    const alerts = this.checkTelemetryAlerts(droneId, telemetry);
    if (alerts.length > 0) {
      await this.handleAlerts(droneId, alerts);
    }

    // Broadcast to subscribers
    this.io.to(`drone:${droneId}`).emit('drone:telemetry_update', {
      droneId,
      telemetry,
      timestamp: new Date()
    });

    // Update fleet statistics
    await this.updateFleetStatistics();
  }

  checkTelemetryAlerts(droneId, telemetry) {
    const alerts = [];

    // Battery level alerts
    if (telemetry.batteryLevel < 20) {
      alerts.push({
        type: 'battery_low',
        severity: telemetry.batteryLevel < 10 ? 'critical' : 'warning',
        message: `Battery level: ${telemetry.batteryLevel}%`,
        droneId,
        timestamp: new Date()
      });
    }

    // Connection alerts
    if (telemetry.signalStrength < 30) {
      alerts.push({
        type: 'signal_weak',
        severity: 'warning',
        message: `Weak signal: ${telemetry.signalStrength}%`,
        droneId,
        timestamp: new Date()
      });
    }

    // Altitude alerts
    if (telemetry.altitude > 120) { // Above legal limit
      alerts.push({
        type: 'altitude_violation',
        severity: 'critical',
        message: `Altitude violation: ${telemetry.altitude}m`,
        droneId,
        timestamp: new Date()
      });
    }

    // Geofence alerts
    if (this.isOutsideGeofence(telemetry.latitude, telemetry.longitude)) {
      alerts.push({
        type: 'geofence_violation',
        severity: 'critical',
        message: 'Drone outside allowed area',
        droneId,
        timestamp: new Date()
      });
    }

    return alerts;
  }

  async handleAlerts(droneId, alerts) {
    for (const alert of alerts) {
      // Store alert in database
      await this.storeAlert(alert);

      // Send to all users monitoring this drone
      this.io.to(`drone:${droneId}`).emit('drone:alert', alert);

      // Send to fleet monitors
      this.io.to('fleet:updates').emit('fleet:alert', alert);

      // For critical alerts, send to all admin users
      if (alert.severity === 'critical') {
        this.io.emit('system:critical_alert', alert);
      }
    }
  }

  async sendFleetOverview(socket) {
    const fleetData = {
      totalDrones: this.activeDrones.size,
      activeDrones: Array.from(this.activeDrones.values()).filter(
        drone => Date.now() - drone.lastUpdate < 60000 // Active in last minute
      ).length,
      alerts: await this.getActiveAlerts(),
      averageBattery: this.calculateAverageBattery(),
      dronesOnMission: await this.getDronesOnMission()
    };

    socket.emit('fleet:overview', fleetData);
  }

  calculateAverageBattery() {
    if (this.activeDrones.size === 0) return 0;
    
    const totalBattery = Array.from(this.activeDrones.values())
      .reduce((sum, drone) => sum + (drone.batteryLevel || 0), 0);
    
    return Math.round(totalBattery / this.activeDrones.size);
  }
}

export default WebSocketServer;
```

```jsx
// components/LiveDashboard/LiveDashboard.jsx
import React, { useState, useEffect } from 'react';
import { useWebSocket } from '../../hooks/useWebSocket';
import FleetMap from './FleetMap/FleetMap';
import FleetStats from './FleetOverview/FleetStats';
import DroneDetailPanel from './DroneDetails/DroneDetailPanel';
import AlertsPanel from './AlertsPanel/AlertsPanel';

const LiveDashboard = () => {
  const [selectedDrone, setSelectedDrone] = useState(null);
  const [fleetData, setFleetData] = useState({});
  const [alerts, setAlerts] = useState([]);
  const [activeDrones, setActiveDrones] = useState(new Map());

  const { socket, connected } = useWebSocket();

  useEffect(() => {
    if (!socket) return;

    // Subscribe to fleet updates
    socket.emit('fleet:subscribe');

    // Fleet overview updates
    socket.on('fleet:overview', (data) => {
      setFleetData(data);
    });

    // Drone telemetry updates
    socket.on('drone:telemetry_update', (data) => {
      setActiveDrones(prev => {
        const newMap = new Map(prev);
        newMap.set(data.droneId, {
          ...newMap.get(data.droneId),
          ...data.telemetry,
          lastUpdate: new Date()
        });
        return newMap;
      });
    });

    // Alert updates
    socket.on('drone:alert', (alert) => {
      setAlerts(prev => [alert, ...prev].slice(0, 50)); // Keep last 50 alerts
    });

    // Fleet alerts
    socket.on('fleet:alert', (alert) => {
      setAlerts(prev => [alert, ...prev].slice(0, 50));
    });

    return () => {
      socket.off('fleet:overview');
      socket.off('drone:telemetry_update');
      socket.off('drone:alert');
      socket.off('fleet:alert');
    };
  }, [socket]);

  const handleDroneSelect = (droneId) => {
    setSelectedDrone(droneId);
    if (socket && droneId) {
      socket.emit('drone:subscribe', droneId);
    }
  };

  const handleDroneDeselect = () => {
    if (socket && selectedDrone) {
      socket.emit('drone:unsubscribe', selectedDrone);
    }
    setSelectedDrone(null);
  };

  const handleEmergencyStop = (droneId) => {
    if (socket) {
      socket.emit('drone:emergency_stop', droneId);
    }
  };

  return (
    <div className="live-dashboard">
      {/* Connection Status */}
      <div className="connection-status">
        <span className={`status-indicator ${connected ? 'connected' : 'disconnected'}`}>
          {connected ? '🟢 Connected' : '🔴 Disconnected'}
        </span>
      </div>

      {/* Main Grid Layout */}
      <div className="dashboard-grid">
        {/* Fleet Statistics */}
        <div className="fleet-stats-section">
          <FleetStats 
            fleetData={fleetData} 
            activeDrones={activeDrones}
            alerts={alerts.filter(a => a.status === 'active')}
          />
        </div>

        {/* Fleet Map */}
        <div className="map-section">
          <FleetMap
            drones={Array.from(activeDrones.values())}
            selectedDrone={selectedDrone}
            onDroneSelect={handleDroneSelect}
            onDroneDeselect={handleDroneDeselect}
          />
        </div>

        {/* Drone Details */}
        {selectedDrone && (
          <div className="drone-details-section">
            <DroneDetailPanel
              droneId={selectedDrone}
              telemetry={activeDrones.get(selectedDrone)}
              onEmergencyStop={handleEmergencyStop}
              onClose={handleDroneDeselect}
            />
          </div>
        )}

        {/* Alerts Panel */}
        <div className="alerts-section">
          <AlertsPanel
            alerts={alerts}
            onAcknowledge={(alertId) => {
              if (socket) {
                socket.emit('alert:acknowledge', alertId);
              }
            }}
          />
        </div>
      </div>
    </div>
  );
};

export default LiveDashboard;
```

```jsx
// components/LiveDashboard/FleetMap/FleetMap.jsx
import React, { useRef, useEffect } from 'react';
import { MapContainer, TileLayer, Marker, Popup, Polyline } from 'react-leaflet';
import L from 'leaflet';
import DroneMarker from './DroneMarker';
import DroneTrail from './DroneTrail';

const FleetMap = ({ drones, selectedDrone, onDroneSelect, onDroneDeselect }) => {
  const mapRef = useRef();

  // Create custom icons for different drone states
  const createDroneIcon = (drone) => {
    const color = getDroneColor(drone);
    const size = selectedDrone === drone.id ? 35 : 25;
    
    return L.divIcon({
      html: `
        <div class="drone-marker" style="
          width: ${size}px; 
          height: ${size}px; 
          background: ${color}; 
          border-radius: 50%;
          border: 2px solid white;
          display: flex;
          align-items: center;
          justify-content: center;
          font-size: 12px;
          box-shadow: 0 2px 5px rgba(0,0,0,0.3);
        ">
          🚁
        </div>
      `,
      className: 'custom-drone-icon',
      iconSize: [size, size],
      iconAnchor: [size/2, size/2]
    });
  };

  const getDroneColor = (drone) => {
    if (!drone.inAir) return '#gray';
    if (drone.batteryLevel < 20) return '#ff4444';
    if (drone.batteryLevel < 50) return '#ff9800';
    return '#4caf50';
  };

  const handleMapClick = () => {
    if (selectedDrone) {
      onDroneDeselect();
    }
  };

  // Auto-fit map bounds to show all drones
  useEffect(() => {
    if (drones.length > 0 && mapRef.current) {
      const bounds = L.latLngBounds(
        drones.map(drone => [
          drone.location?.coordinates[1], 
          drone.location?.coordinates[0]
        ]).filter(coord => coord[0] && coord[1])
      );
      
      if (bounds.isValid()) {
        mapRef.current.fitBounds(bounds, { padding: [20, 20] });
      }
    }
  }, [drones]);

  return (
    <div className="fleet-map-container">
      <div className="map-header">
        <h3>Fleet Map</h3>
        <div className="map-stats">
          <span>{drones.filter(d => d.inAir).length} Flying</span>
          <span>{drones.filter(d => !d.inAir).length} Landed</span>
        </div>
      </div>
      
      <MapContainer
        ref={mapRef}
        center={[50.4501, 30.5234]} // Default center (Kyiv)
        zoom={10}
        style={{ height: '500px', width: '100%' }}
        onClick={handleMapClick}
      >
        <TileLayer
          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>'
          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
        />

        {/* Drone Markers */}
        {drones.map(drone => {
          if (!drone.location?.coordinates) return null;
          
          const [lng, lat] = drone.location.coordinates;
          
          return (
            <Marker
              key={drone.id}
              position={[lat, lng]}
              icon={createDroneIcon(drone)}
              eventHandlers={{
                click: () => onDroneSelect(drone.id)
              }}
            >
              <Popup>
                <div className="drone-popup">
                  <h4>{drone.name || `Drone ${drone.id}`}</h4>
                  <div className="popup-stats">
                    <div>Battery: {drone.batteryLevel}%</div>
                    <div>Altitude: {drone.altitude}m</div>
                    <div>Speed: {drone.speed}m/s</div>
```

```javascript
// src/services/analytics/realTimeAnalytics.js
import LiveTelemetry from '../../models/liveTelemetry.model.js';
import Alert from '../../models/alert.model.js';
import Redis from 'ioredis';

class RealTimeAnalytics {
  constructor() {
    this.redis = new Redis({
      host: process.env.REDIS_HOST || 'localhost',
      port: process.env.REDIS_PORT || 6379
    });
    
    this.metrics = new Map();
    this.startPeriodicAnalysis();
  }

  async processTelemetryUpdate(droneId, telemetry) {
    // Store in time-series for analytics
    await this.storeTelemetryMetrics(droneId, telemetry);
    
    // Update real-time metrics
    await this.updateRealtimeMetrics(droneId, telemetry);
    
    // Detect anomalies
    const anomalies = await this.detectAnomalies(droneId, telemetry);
    if (anomalies.length > 0) {
      await this.handleAnomalies(droneId, anomalies);
    }
    
    // Update fleet-wide statistics
    await this.updateFleetStatistics();
  }

  async storeTelemetryMetrics(droneId, telemetry) {
    const telemetryRecord = new LiveTelemetry({
      droneId,
      location: {
        type: 'Point',
        coordinates: [telemetry.longitude, telemetry.latitude]
      },
      altitude: telemetry.altitude,
      heading: telemetry.heading,
      speed: telemetry.speed,
      batteryLevel: telemetry.batteryLevel,
      batteryVoltage: telemetry.batteryVoltage,
      signalStrength: telemetry.signalStrength,
      flightMode: telemetry.flightMode,
      armed: telemetry.armed,
      inAir: telemetry.inAir,
      timestamp: new Date()
    });

    await telemetryRecord.save();
  }

  async updateRealtimeMetrics(droneId, telemetry) {
    const metrics = {
      batteryDrainRate: await this.calculateBatteryDrainRate(droneId, telemetry.batteryLevel),
      flightEfficiency: this.calculateFlightEfficiency(telemetry),
      signalQuality: this.assessSignalQuality(telemetry.signalStrength),
      stabilityScore: this.calculateStabilityScore(telemetry),
      riskAssessment: await this.assessFlightRisk(droneId, telemetry)
    };

    // Store in Redis for real-time access
    await this.redis.setex(
      `metrics:${droneId}`,
      300, // 5 minutes TTL
      JSON.stringify(metrics)
    );

    this.metrics.set(droneId, metrics);
  }

  async calculateBatteryDrainRate(droneId, currentBattery) {
    // Get battery history from last 5 minutes
    const fiveMinutesAgo = new Date(Date.now() - 5 * 60000);
    const batteryHistory = await LiveTelemetry
      .find({
        droneId,
        timestamp: { $gte: fiveMinutesAgo }
      })
      .sort({ timestamp: 1 })
      .select('batteryLevel timestamp');

    if (batteryHistory.length < 2) return 0;

    const first = batteryHistory[0];
    const last = batteryHistory[batteryHistory.length - 1];
    const timeDiff = (last.timestamp - first.timestamp) / 1000 / 60; // minutes
    const batteryDiff = first.batteryLevel - last.batteryLevel;

    return timeDiff > 0 ? batteryDiff / timeDiff : 0; // %/minute
  }

  calculateFlightEfficiency(telemetry) {
    // Simple efficiency calculation based on speed and battery usage
    const speedEfficiency = Math.min(telemetry.speed / 15, 1); // Optimal speed ~15 m/s
    const batteryEfficiency = telemetry.batteryLevel / 100;
    const signalEfficiency = telemetry.signalStrength / 100;

    return (speedEfficiency + batteryEfficiency + signalEfficiency) / 3;
  }

  calculateStabilityScore(telemetry) {
    let score = 100;

    // Reduce score based on vibrations and erratic movements
    if (telemetry.verticalSpeed && Math.abs(telemetry.verticalSpeed) > 5) {
      score -= 20; // High vertical speed indicates instability
    }

    if (telemetry.accelerometer) {
      const totalAccel = Math.sqrt(
        telemetry.accelerometer.x ** 2 +
        telemetry.accelerometer.y ** 2 +
        telemetry.accelerometer.z ** 2
      );
      if (totalAccel > 15) score -= 30; // High acceleration
    }

    return Math.max(0, score);
  }

  async detectAnomalies(droneId, telemetry) {
    const anomalies = [];

    // Get historical baseline
    const baseline = await this.getBaselineMetrics(droneId);
    
    // Battery anomaly detection
    if (telemetry.batteryLevel < baseline.avgBatteryLevel - (2 * baseline.batteryStdDev)) {
      anomalies.push({
        type: 'battery_anomaly',
        severity: 'warning',
        description: 'Battery level significantly below normal',
        value: telemetry.batteryLevel,
        expected: baseline.avgBatteryLevel
      });
    }

    // Speed anomaly detection
    if (telemetry.speed > baseline.avgSpeed + (3 * baseline.speedStdDev)) {
      anomalies.push({
        type: 'speed_anomaly',
        severity: 'warning',
        description: 'Unusual speed detected',
        value: telemetry.speed,
        expected: baseline.avgSpeed
      });
    }

    // Signal strength anomaly
    if (telemetry.signalStrength < baseline.avgSignal - (2 * baseline.signalStdDev)) {
      anomalies.push({
        type: 'signal_anomaly',
        severity: 'warning',
        description: 'Signal strength below normal',
        value: telemetry.signalStrength,
        expected: baseline.avgSignal
      });
    }

    return anomalies;
  }

  async getBaselineMetrics(droneId) {
    const oneWeekAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
    
    const pipeline = [
      {
        $match: {
          droneId: droneId,
          timestamp: { $gte: oneWeekAgo },
          inAir: true // Only consider flight data
        }
      },
      {
        $group: {
          _id: null,
          avgBatteryLevel: { $avg: '$batteryLevel' },
          avgSpeed: { $avg: '$speed' },
          avgSignal: { $avg: '$signalStrength' },
          batteryLevels: { $push: '$batteryLevel' },
          speeds: { $push: '$speed' },
          signals: { $push: '$signalStrength' }
        }
      }
    ];

    const result = await LiveTelemetry.aggregate(pipeline);
    if (!result.length) return this.getDefaultBaseline();

    const data = result[0];
    
    return {
      avgBatteryLevel: data.avgBatteryLevel,
      avgSpeed: data.avgSpeed,
      avgSignal: data.avgSignal,
      batteryStdDev: this.calculateStandardDeviation(data.batteryLevels),
      speedStdDev: this.calculateStandardDeviation(data.speeds),
      signalStdDev: this.calculateStandardDeviation(data.signals)
    };
  }

  calculateStandardDeviation(values) {
    const mean = values.reduce((sum, val) => sum + val, 0) / values.length;
    const variance = values.reduce((sum, val) => sum + (val - mean) ** 2, 0) / values.length;
    return Math.sqrt(variance);
  }

  startPeriodicAnalysis() {
    // Run fleet analysis every minute
    setInterval(async () => {
      await this.analyzeFleetPerformance();
      await this.cleanupOldData();
    }, 60000);
  }

  async analyzeFleetPerformance() {
    const activeDrones = await this.getActiveDrones();
    
    const fleetMetrics = {
      totalDrones: activeDrones.length,
      averageBattery: this.calculateAverageBattery(activeDrones),
      averageSignal: this.calculateAverageSignal(activeDrones),
      dronesInAir: activeDrones.filter(d => d.inAir).length,
      criticalBatteryCount: activeDrones.filter(d => d.batteryLevel < 20).length,
      weakSignalCount: activeDrones.filter(d => d.signalStrength < 30).length,
      timestamp: new Date()
    };

    // Store fleet metrics
    await this.redis.setex(
      'fleet:metrics',
      300,
      JSON.stringify(fleetMetrics)
    );
  }

  async cleanupOldData() {
    // Remove telemetry data older than 24 hours
    const oneDayAgo = new Date(Date.now() - 24 * 60 * 60 * 1000);
    await LiveTelemetry.deleteMany({ timestamp: { $lt: oneDayAgo } });
  }
}
```

```javascript
// src/services/alertManager.js
import Alert from '../models/alert.model.js';
import nodemailer from 'nodemailer';

class AlertManager {
  constructor(io) {
    this.io = io;
    this.alertRules = new Map();
    this.setupDefaultRules();
    this.emailTransporter = this.setupEmailTransporter();
  }

  setupDefaultRules() {
    // Battery alerts
    this.alertRules.set('battery_low', {
      condition: (telemetry) => telemetry.batteryLevel < 30,
      severity: 'warning',
      title: 'Low Battery',
      autoActions: ['notify_operator'],
      cooldown: 300000 // 5 minutes
    });

    this.alertRules.set('battery_critical', {
      condition: (telemetry) => telemetry.batteryLevel < 15,
      severity: 'critical',
      title: 'Critical Battery Level',
      autoActions: ['notify_operator', 'suggest_rtl'],
      cooldown: 60000 // 1 minute
    });

    // Signal alerts
    this.alertRules.set('signal_weak', {
      condition: (telemetry) => telemetry.signalStrength < 30,
      severity: 'warning',
      title: 'Weak Signal',
      autoActions: ['notify_operator'],
      cooldown: 120000 // 2 minutes
    });

    // Altitude alerts
    this.alertRules.set('altitude_violation', {
      condition: (telemetry) => telemetry.altitude > 120, // Legal limit
      severity: 'critical',
      title: 'Altitude Violation',
      autoActions: ['notify_operator', 'log_violation'],
      cooldown: 30000 // 30 seconds
    });
  }

  async processAlerts(droneId, telemetry) {
    const triggeredAlerts = [];

    for (const [alertType, rule] of this.alertRules) {
      if (rule.condition(telemetry)) {
        // Check cooldown
        if (await this.isInCooldown(droneId, alertType, rule.cooldown)) {
          continue;
        }

        const alert = await this.createAlert(droneId, alertType, rule, telemetry);
        triggeredAlerts.push(alert);

        // Execute auto actions
        await this.executeAutoActions(alert, rule.autoActions);

        // Update cooldown
        await this.updateCooldown(droneId, alertType);
      }
    }

    return triggeredAlerts;
  }

  async createAlert(droneId, type, rule, telemetry) {
    const alert = new Alert({
      droneId,
      userId: telemetry.userId,
      type,
      severity: rule.severity,
      title: rule.title,
      message: this.generateMessage(type, rule, telemetry),
      telemetrySnapshot: telemetry,
      location: {
        type: 'Point',
        coordinates: [telemetry.longitude, telemetry.latitude]
      }
    });

    await alert.save();

    // Emit to WebSocket subscribers
    this.io.to(`drone:${droneId}`).emit('drone:alert', alert);
    this.io.to('fleet:updates').emit('fleet:alert', alert);

    // Send critical alerts to all users
    if (rule.severity === 'critical') {
      this.io.emit('system:critical_alert', alert);
    }

    return alert;
  }

  generateMessage(type, rule, telemetry) {
    switch (type) {
      case 'battery_low':
        return `Battery level is ${telemetry.batteryLevel}%. Consider returning to base.`;
      case 'battery_critical':
        return `Critical battery level: ${telemetry.batteryLevel}%. Immediate return to base recommended.`;
      case 'signal_weak':
        return `Signal strength is ${telemetry.signalStrength}%. Connection may be unstable.`;
      case 'altitude_violation':
        return `Drone altitude ${telemetry.altitude}m exceeds legal limit of 120m.`;
      default:
        return rule.title;
    }
  }

  async executeAutoActions(alert, actions) {
    for (const action of actions) {
      try {
        await this.executeAction(alert, action);
        
        // Log successful action
        alert.autoActions.push({
          action,
          executed: true,
          executedAt: new Date(),
          result: 'success'
        });
      } catch (error) {
        // Log failed action
        alert.autoActions.push({
          action,
          executed: false,
          executedAt: new Date(),
          result: error.message
        });
      }
    }

    await alert.save();
  }

  async executeAction(alert, action) {
    switch (action) {
      case 'notify_operator':
        await this.notifyOperator(alert);
        break;
        
      case 'suggest_rtl':
        await this.suggestReturnToLaunch(alert);
        break;
        
      case 'log_violation':
        await this.logViolation(alert);
        break;
        
      case 'emergency_land':
        await this.initiateEmergencyLanding(alert);
        break;
    }
  }

  async notifyOperator(alert) {
    // Send email notification
    if (this.emailTransporter) {
      const mailOptions = {
        from: process.env.ALERT_EMAIL_FROM,
        to: process.env.OPERATOR_EMAIL,
        subject: `Drone Alert: ${alert.title}`,
        html: `
          <h2>Drone Alert</h2>
          <p><strong>Drone ID:</strong> ${alert.droneId}</p>
          <p><strong>Alert Type:</strong> ${alert.type}</p>
          <p><strong>Severity:</strong> ${alert.severity}</p>
          <p><strong>Message:</strong> ${alert.message}</p>
          <p><strong>Time:</strong> ${alert.createdAt}</p>
          <p><strong>Location:</strong> ${alert.location.coordinates[1]}, ${alert.location.coordinates[0]}</p>
        `
      };

      await this.emailTransporter.sendMail(mailOptions);
    }

    // Send SMS notification for critical alerts
    if (alert.severity === 'critical') {
      await this.sendSMSNotification(alert);
    }
  }

  async isInCooldown(droneId, alertType, cooldownMs) {
    const lastAlert = await Alert.findOne({
      droneId,
      type: alertType,
      createdAt: { $gte: new Date(Date.now() - cooldownMs) }
    }).sort({ createdAt: -1 });

    return !!lastAlert;
  }

  async acknowledgeAlert(alertId, userId) {
    const alert = await Alert.findByIdAndUpdate(
      alertId,
      {
        status: 'acknowledged',
        acknowledgedBy: userId,
        acknowledgedAt: new Date()
      },
      { new: true }
    );

    if (alert) {
      this.io.emit('alert:acknowledged', {
        alertId,
        acknowledgedBy: userId,
        acknowledgedAt: new Date()
      });
    }

    return alert;
  }

  async resolveAlert(alertId, userId) {
    const alert = await Alert.findByIdAndUpdate(
      alertId,
      {
        status: 'resolved',
        resolvedAt: new Date()
      },
      { new: true }
    );

    if (alert) {
      this.io.emit('alert:resolved', {
        alertId,
        resolvedAt: new Date()
      });
    }

    return alert;
  }

  setupEmailTransporter() {
    if (!process.env.SMTP_HOST) return null;

    return nodemailer.createTransporter({
      host: process.env.SMTP_HOST,
      port: process.env.SMTP_PORT || 587,
      secure: false,
      auth: {
        user: process.env.SMTP_USER,
        pass: process.env.SMTP_PASS
      }
    });
  }
}
```

```javascript
    const connections = [];
    const startTime = performance.now();
    
    for (let i = 0; i < 100; i++) {
      const client = Client('http://localhost:3000/api/ws');
      connections.push(client);
      
      client.emit('fleet:subscribe');
    }
    
    // Send telemetry from 50 drones
    for (let i = 0; i < 50; i++) {
      wsServer.processTelemetry({
        droneId: `drone-${i}`,
        batteryLevel: 75,
        altitude: 50
      }, 'test-user');
    }
    
    const endTime = performance.now();
    expect(endTime - startTime).toBeLessThan(2000);
    
    connections.forEach(client => client.close());
  });
});
```
